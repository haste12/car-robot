About Our Project

As five Computer Engineering students at the Lebanese French University, we designed and built Car Robot — a fully autonomous smart car robot that integrates multiple intelligent systems to navigate, respond to environment input, and be controlled remotely.

The car uses an Arduino Uno microcontroller as its brain, combined with ultrasonic, infrared, and sound sensors, a Bluetooth module, a sound sensor module, and a servo-driven steering mechanism.

This project was completed under the supervision of Ms. Rawshan Nuree Othman, to whom we extend our deepest gratitude for her guidance and continuous support.

01

Obstacle Avoidance

The robot uses an HC-SR04 ultrasonic sensor mounted on a servo motor to scan the environment 180°. When an obstacle is detected within 20 cm, the car stops, evaluates left and right distances, and chooses the clearest path.

  • HC-SR04 Ultrasonic Sensor
  • SG90 Servo (180° scan)
  • L298N Motor Driver
  • Real-time distance calculation

#include <Servo.h>
#include <NewPing.h>

#define TRIG_PIN   12
#define ECHO_PIN   11
#define MAX_DIST   200
#define SAFE_DIST  20

NewPing sonar(TRIG_PIN, ECHO_PIN, MAX_DIST);
Servo scanServo;

int m1 = 2, m2 = 3, m3 = 4, m4 = 5;
int enA = 9, enB = 10;

void setup() {
  scanServo.attach(6);
  for (int p : {m1,m2,m3,m4,enA,enB})
    pinMode(p, OUTPUT);
  analogWrite(enA, 180);
  analogWrite(enB, 180);
}

void loop() {
  int dist = sonar.ping_cm();
  if (dist > 0 && dist < SAFE_DIST) {
    stopCar();
    delay(300);
    int leftD  = scanDir(150);
    int rightD = scanDir(30);
    (leftD > rightD) ? turnLeft() : turnRight();
    delay(500);
  } else {
    moveForward();
  }
}

int scanDir(int angle) {
  scanServo.write(angle);
  delay(300);
  return sonar.ping_cm();
}

void moveForward()  { digitalWrite(m1,HIGH);digitalWrite(m2,LOW); digitalWrite(m3,HIGH);digitalWrite(m4,LOW); }
void turnLeft()     { digitalWrite(m1,LOW); digitalWrite(m2,HIGH);digitalWrite(m3,HIGH);digitalWrite(m4,LOW); }
void turnRight()    { digitalWrite(m1,HIGH);digitalWrite(m2,LOW); digitalWrite(m3,LOW); digitalWrite(m4,HIGH); }
void stopCar()      { digitalWrite(m1,LOW); digitalWrite(m2,LOW); digitalWrite(m3,LOW); digitalWrite(m4,LOW); }

      case '-': spd = max(spd-20, 80);
               analogWrite(enA, spd);
               analogWrite(enB, spd); break;
    }
  }
}

Line Following

Two IR sensors positioned at the front-bottom of the car detect a black line on a white surface. The car adjusts its direction in real time — following curves, corners, and straight paths autonomously.

  • 2× IR Reflective Sensors (digital)
  • PID-like correction logic
  • Handles sharp corners
  • Speed adjustment on turns

Arduino C++ 
#define IR_LEFT   7
#define IR_RIGHT  8
#define M1  2
#define M2  3
#define M3  4
#define M4  5
#define ENA 9
#define ENB 10

#define BASE_SPD  160
#define TURN_SPD  220

void setup() {
  pinMode(IR_LEFT,  INPUT);
  pinMode(IR_RIGHT, INPUT);
  for (int p : {M1,M2,M3,M4,ENA,ENB})
    pinMode(p, OUTPUT);
}

void loop() {
  bool L = digitalRead(IR_LEFT);  // LOW = on line
  bool R = digitalRead(IR_RIGHT);

  if (!L && !R) {          // both on line → forward
    analogWrite(ENA, BASE_SPD);
    analogWrite(ENB, BASE_SPD);
    forward();
  } else if (L && !R) {    // left off → turn left
    analogWrite(ENA, TURN_SPD);
    analogWrite(ENB, BASE_SPD/2);
    turnLeft();
  } else if (!L && R) {   // right off → turn right
    analogWrite(ENA, BASE_SPD/2);
    analogWrite(ENB, TURN_SPD);
    turnRight();
  } else {               // both off → stop
    stopCar();
  }
}

void forward()  { digitalWrite(M1,HIGH);digitalWrite(M2,LOW); digitalWrite(M3,HIGH);digitalWrite(M4,LOW); }
void turnLeft()  { digitalWrite(M1,LOW); digitalWrite(M2,HIGH);digitalWrite(M3,HIGH);digitalWrite(M4,LOW); }
void turnRight() { digitalWrite(M1,HIGH);digitalWrite(M2,LOW); digitalWrite(M3,LOW); digitalWrite(M4,HIGH); }
void stopCar()   { digitalWrite(M1,LOW); digitalWrite(M2,LOW); digitalWrite(M3,LOW); digitalWrite(M4,LOW); }

Sound Controller

A KY-038 sound sensor detects acoustic signals such as claps or loud sounds. A double-clap pattern toggles the motors on/off in real time, allowing hands-free control without any wireless hardware.

  • KY-038 Sound Sensor (analog + digital)
  • Double-clap toggle logic
  • Debounce & timing window (600 ms)
  • LED status indicator

Arduino C++ 
#define SOUND_DIG  7   // digital threshold pin
#define SOUND_ANA  A0  // analog level pin
#define LED_PIN    13
#define M1  2
#define M2  3
#define M3  4
#define M4  5
#define ENA 9
#define ENB 10

bool motorState = false;
int  clapCount  = 0;
unsigned long lastClap = 0;

void setup() {
  Serial.begin(9600);
  pinMode(SOUND_DIG, INPUT);
  pinMode(LED_PIN,   OUTPUT);
  for (int p : {M1,M2,M3,M4,ENA,ENB})
    pinMode(p, OUTPUT);
  analogWrite(ENA, 180);
  analogWrite(ENB, 180);
}

void loop() {
  if (digitalRead(SOUND_DIG) == LOW) {  // sound detected
    unsigned long now = millis();
    if (now - lastClap < 600) {         // double-clap window
      clapCount++;
    } else {
      clapCount = 1;
    }
    lastClap = now;
    delay(80);                           // debounce

    if (clapCount >= 2) {
      clapCount  = 0;
      motorState = !motorState;            // toggle motors
      motorState ? moveForward() : stopCar();
      digitalWrite(LED_PIN, motorState);
    }
  }
  Serial.println(analogRead(SOUND_ANA));  // monitor level
}

void moveForward() {
  digitalWrite(M1,HIGH); digitalWrite(M2,LOW);
  digitalWrite(M3,HIGH); digitalWrite(M4,LOW);
}
void stopCar() {
  digitalWrite(M1,LOW); digitalWrite(M2,LOW);
  digitalWrite(M3,LOW); digitalWrite(M4,LOW);
}

Voice Control

A DFRobot Voice Recognition module listens for pre-trained voice commands in real time. Saying "Forward", "Stop", "Left", or "Right" drives the car hands-free — no phone or computer needed.

  • DFRobot Voice Recognition V3
  • 5 customizable voice commands
  • UART communication (9600 baud)
  • Speaker-independent recognition

#include <SoftwareSerial.h>

SoftwareSerial voice(10, 11); // RX=10, TX=11

// Command byte codes (trained in DFRobot app)
#define CMD_FORWARD  0x11
#define CMD_BACKWARD 0x12
#define CMD_LEFT     0x13
#define CMD_RIGHT    0x14
#define CMD_STOP     0x15

int m1=2,m2=3,m3=4,m4=5;

void setup() {
  voice.begin(9600);
  for (int p : {m1,m2,m3,m4})
    pinMode(p, OUTPUT);
  addCommand(CMD_FORWARD,  "forward");
  addCommand(CMD_BACKWARD, "backward");
  addCommand(CMD_LEFT,     "turn left");
  addCommand(CMD_RIGHT,    "turn right");
  addCommand(CMD_STOP,     "stop");
}

void loop() {
  if (voice.available()) {
    byte cmd = voice.read();
    switch(cmd) {
      case CMD_FORWARD:  forward();  break;
      case CMD_BACKWARD: backward(); break;
      case CMD_LEFT:     turnLeft(); break;
      case CMD_RIGHT:    turnRight(); break;
      case CMD_STOP:     stopCar();  break;
    }
  }
}

void addCommand(byte id, String label) {
  // DFRobot training protocol
  voice.write(0xAA); voice.write(id);
  delay(100);
}
